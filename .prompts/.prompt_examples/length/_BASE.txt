  constructor
  · rintro ⟨h0, h1⟩
    constructor
    · exact h0
    intro h2
    apply h1
    rw [h2]
  rintro ⟨h0, h1⟩
  constructor
  · exact h0
  intro h2
  apply h1
  apply le_antisymm h0 h2

  · rintro ⟨h0, h1⟩
    exact ⟨h0, fun h2 => h1 (by rw [h2])⟩

    exact ⟨h0, fun h2 => h1 (by linarith)⟩

    apply h
    apply inverse_spec
    use y
  intro h x1 x2 e
  rw [← h x1, ← h x2, e]

theorem C04_S02_24: Injective f ↔ LeftInverse (inverse f) f  := by
  constructor
  · exact fun h y ↦ h (inverse_spec _ ⟨y, rfl⟩)
  · exact fun h x1 x2 e ↦ by rw [←h x1, e, h x2]

    (hs : 0 < s)
    (ht : 0 < t)
    (hu : 0 < u)
    (h : s^4 - t^4 = u^2) : False := by

  lift s to ℕ using Int.le_of_lt hs
  lift t to ℕ using Int.le_of_lt ht
  lift u to ℕ using Int.le_of_lt hu
  replace hs : 0 < s := Int.ofNat_pos.mp hs
  replace ht : 0 < t := Int.ofNat_pos.mp ht
  replace hy : 0 < u := Int.ofNat_pos.mp hu
  have h' : s ^ 4 = t ^ 4 + u ^ 2 := by exact Int.ofNat_inj.mp h
  exact lemma_1' s t u hs ht hy h'
